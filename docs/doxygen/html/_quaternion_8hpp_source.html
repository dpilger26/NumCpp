220 // Method Description:
229 static NdArray<double> angularVelocity(const Quaternion& inQuat1, const Quaternion& inQuat2, double inTime)
230 {
231 const auto q0 = inQuat1.toNdArray();
232 const auto q1 = inQuat2.toNdArray();
233
234 auto qDot = q1 - q0;
235 qDot /= inTime;
236
237 auto eyeTimesScalar = NdArray<double>(3);
238 eyeTimesScalar.zeros();
239 eyeTimesScalar(0, 0) = inQuat2.s();
240 eyeTimesScalar(1, 1) = inQuat2.s();
241 eyeTimesScalar(2, 2) = inQuat2.s();
242
243 const auto epsilonHat = linalg::hat<double>(inQuat2.i(), inQuat2.j(), inQuat2.k());
244 auto q = NdArray<double>(4, 3);
245 q.put(Slice(0, 3), Slice(0, 3), eyeTimesScalar + epsilonHat);
246 q(3, 0) = -inQuat2.i();
247 q(3, 1) = -inQuat2.j();
248 q(3, 2) = -inQuat2.k();
249
250 auto omega = q.transpose().dot(qDot.transpose());
251 return omega *= 2.;
252 }

378
379 stl_algorithms::transform(inQuat1.components_.begin(),
380 inQuat1.components_.end(),
381 inQuat2.components_.begin(),
382 newComponents.begin(),
383 [inPercent, oneMinus](double component1, double component2) -> double
384 { return oneMinus * component1 + inPercent * component2; });
385
386 return { newComponents };
387 }

726 static Quaternion slerp(const Quaternion& inQuat1, const Quaternion& inQuat2, double inPercent)
727 {
728 if (inPercent < 0 || inPercent > 1)
729 {
730 THROW_INVALID_ARGUMENT_ERROR("input percent must be of the range [0, 1]");
731 }
732
733 if (utils::essentiallyEqual(inPercent, 0.))
734 {
735 return inQuat1;
736 }
737 if (utils::essentiallyEqual(inPercent, 1.))
738 {
739 return inQuat2;
740 }
741
742 double dotProduct = dot<double>(inQuat1.toNdArray(), inQuat2.toNdArray()).item();
743
744 // If the dot product is negative, the quaternions
745 // have opposite handed-ness and slerp won't take
746 // the shorter path. Fix by reversing one quaternion.
747 auto quat1Copy = Quaternion(inQuat1);
748 if (dotProduct < 0.)
749 {
750 quat1Copy *= -1.;
751 dotProduct *= -1.;
752 }
753
754 constexpr auto DOT_THRESHOLD = 0.9995;
755 if (dotProduct > DOT_THRESHOLD)
756 {
757 // If the inputs are too close for comfort, linearly interpolate
758 // and normalize the result.
759 return nlerp(inQuat1, inQuat2, inPercent);
760 }
761
762 dotProduct = clip(dotProduct, -1., 1.); // Robustness: Stay within domain of acos()
763 const auto theta0 = std::acos(dotProduct); // angle between input vectors
764 const auto theta = theta0 * inPercent; // angle between v0 and result
765
766 const auto s0 = std::cos(theta) -
767 dotProduct * std::sin(theta) / std::sin(theta0); // == sin(theta_0 - theta) / sin(theta_0)
768 const auto s1 = std::sin(theta) / std::sin(theta0);
769
770 const auto interpQuat = (quat1Copy.toNdArray() * s0) + (inQuat2.toNdArray() * s1);
771 return Quaternion(interpQuat); // NOLINT(modernize-return-braced-init-list)
772 }

807 [[nodiscard]] NdArray<double> toDCM() const
808 {
809 auto dcm = NdArray<double>(3);
810
811 const auto q0 = i();
812 const auto q1 = j();
813 const auto q2 = k();
814 const auto q3 = s();
815
816 const auto q0sqr = utils::sqr(q0);
817 const auto q1sqr = utils::sqr(q1);
818 const auto q2sqr = utils::sqr(q2);
819 const auto q3sqr = utils::sqr(q3);
820
821 dcm(0, 0) = q3sqr + q0sqr - q1sqr - q2sqr;
822 dcm(0, 1) = 2. * (q0 * q1 - q3 * q2);
823 dcm(0, 2) = 2. * (q0 * q2 + q3 * q1);
824 dcm(1, 0) = 2. * (q0 * q1 + q3 * q2);
825 dcm(1, 1) = q3sqr + q1sqr - q0sqr - q2sqr;
826 dcm(1, 2) = 2. * (q1 * q2 - q3 * q0);
827 dcm(2, 0) = 2. * (q0 * q2 - q3 * q1);
828 dcm(2, 1) = 2. * (q1 * q2 + q3 * q0);
829 dcm(2, 2) = q3sqr + q2sqr - q0sqr - q1sqr;
830
831 return dcm;
832 }

1242 // Method Description:
1247 void dcmToQuat(const NdArray<double>& dcm)
1248 {
1249 const auto inShape = dcm.shape();
1250 if (!(inShape.rows == 3 && inShape.cols == 3))
1251 {
1252 THROW_INVALID_ARGUMENT_ERROR("input direction cosine matrix must have shape = (3,3).");
1253 }
1254
1255 NdArray<double> checks(1, 4);
1256 checks[0] = 1 + dcm(0, 0) + dcm(1, 1) + dcm(2, 2);
1257 checks[1] = 1 + dcm(0, 0) - dcm(1, 1) - dcm(2, 2);
1258 checks[2] = 1 - dcm(0, 0) + dcm(1, 1) - dcm(2, 2);
1259 checks[3] = 1 - dcm(0, 0) - dcm(1, 1) + dcm(2, 2);
1260
1261 const uint32 maxIdx = argmax(checks).item();
1262
1263 switch (maxIdx)
1264 {
1265 case 0:
1266 {
1267 components_[3] = 0.5 * std::sqrt(1 + dcm(0, 0) + dcm(1, 1) + dcm(2, 2));
1268 components_[0] = (dcm(2, 1) - dcm(1, 2)) / (4 * components_[3]);
1269 components_[1] = (dcm(0, 2) - dcm(2, 0)) / (4 * components_[3]);
1270 components_[2] = (dcm(1, 0) - dcm(0, 1)) / (4 * components_[3]);
1271
1272 break;
1273 }
1274 case 1:
1275 {
1276 components_[0] = 0.5 * std::sqrt(1 + dcm(0, 0) - dcm(1, 1) - dcm(2, 2));
1277 components_[1] = (dcm(1, 0) + dcm(0, 1)) / (4 * components_[0]);
1278 components_[2] = (dcm(2, 0) + dcm(0, 2)) / (4 * components_[0]);
1279 components_[3] = (dcm(2, 1) - dcm(1, 2)) / (4 * components_[0]);
1280
1281 break;
1282 }
1283 case 2:
1284 {
1285 components_[1] = 0.5 * std::sqrt(1 - dcm(0, 0) + dcm(1, 1) - dcm(2, 2));
1286 components_[0] = (dcm(1, 0) + dcm(0, 1)) / (4 * components_[1]);
1287 components_[2] = (dcm(2, 1) + dcm(1, 2)) / (4 * components_[1]);
1288 components_[3] = (dcm(0, 2) - dcm(2, 0)) / (4 * components_[1]);
1289
1290 break;
1291 }
1292 case 3:
1293 {
1294 components_[2] = 0.5 * std::sqrt(1 - dcm(0, 0) - dcm(1, 1) + dcm(2, 2));
1295 components_[0] = (dcm(2, 0) + dcm(0, 2)) / (4 * components_[2]);
1296 components_[1] = (dcm(2, 1) + dcm(1, 2)) / (4 * components_[2]);
1297 components_[3] = (dcm(1, 0) - dcm(0, 1)) / (4 * components_[2]);
1298
1299 break;
1300 }

1305 // Method Description:
1312 void propagate(const Vec3& inAngularVelocity, double inDeltaT, const NdArray<double>& inOmegaOperator)
1313 {
1314 if (utils::essentiallyEqual(inDeltaT, 0.))
1315 {
1316 return;
1317 }
1318
1319 const auto angularVelocityNorm = inAngularVelocity.norm();
1320 if (utils::essentiallyEqual(angularVelocityNorm, 0.))
1321 {
1322 return;
1323 }
1324
1325 const auto halfDeltaT = inDeltaT / 2.;
1326 const auto halfAngle = angularVelocityNorm * halfDeltaT;
1327 const auto sinHalfAngle = std::sin(halfAngle) / angularVelocityNorm;
1328 const auto cosHalfAngle = std::cos(halfAngle);
1329
1330 const auto lhs = cosHalfAngle * eye<double>(4);
1331 const auto rhs = sinHalfAngle * inOmegaOperator;
1332
1333 const auto qDeltaT = (lhs + rhs).dot(toNdArray().transpose());
1334
1335 components_[0] = qDeltaT[0];
1336 components_[1] = qDeltaT[1];
1337 components_[2] = qDeltaT[2];
1338 components_[3] = qDeltaT[3];
1339
1340 normalize();
1341 }
